package kh.edu.istad.platform.iamapplicationservice.service.user;

import kh.edu.istad.platform.iamapplicationservice.command.user.ChangePasswordCommand;
import kh.edu.istad.platform.iamapplicationservice.port.out.CurrentActorProvider;
import kh.edu.istad.platform.iamapplicationservice.port.out.PasswordHasher;
import kh.edu.istad.platform.iamapplicationservice.port.out.UserRepository;
import kh.edu.istad.platform.iamapplicationservice.usecase.user.ChangePasswordUseCase;
import kh.edu.istad.platform.iamdomaincore.model.User;

import java.time.OffsetDateTime;

public class ChangePasswordService implements ChangePasswordUseCase {

    private final UserRepository repository;
    private final PasswordHasher passwordHasher;
    private final CurrentActorProvider actorProvider;

    public ChangePasswordService(
            UserRepository repository,
            PasswordHasher passwordHasher,
            CurrentActorProvider actorProvider
    ) {
        this.repository = repository;
        this.passwordHasher = passwordHasher;
        this.actorProvider = actorProvider;
    }

    @Override
    public void execute(ChangePasswordCommand command) {
        if (command == null) {
            throw new IllegalArgumentException("Change password command is required");
        }

        if (command.currentPassword() == null || command.currentPassword().isBlank()) {
            throw new IllegalArgumentException("Current password is required");
        }

        if (command.newPassword() == null || command.newPassword().isBlank()) {
            throw new IllegalArgumentException("New password is required");
        }

        User user = UserStateSupport.requireActiveUser(
                repository, command.userUuid()
        );

        user.validatePasswordAvailable();

        if (!passwordHasher.matches(
                command.currentPassword(), user.passwordHash()
        )) {
            throw new IllegalArgumentException("Current password is incorrect");
        }

        String passwordHash = passwordHasher.hash(command.newPassword());

        if (passwordHash == null || passwordHash.isBlank()) {
            throw new IllegalStateException("Password hashing failed");
        }

        String actor = actorProvider.getCurrentActor();

        if (actor == null || actor.isBlank()) {
            throw new IllegalStateException("Current actor is required");
        }

        repository.save(UserPasswordSupport.updatePassword(
                user, passwordHash, actor, OffsetDateTime.now()
        ));
    }
}

package kh.edu.istad.platform.iamapplicationservice.service.user;

import kh.edu.istad.platform.iamapplicationservice.command.user.CreateUserCommand;
import kh.edu.istad.platform.iamapplicationservice.port.out.CurrentActorProvider;
import kh.edu.istad.platform.iamapplicationservice.port.out.PasswordHasher;
import kh.edu.istad.platform.iamapplicationservice.port.out.UserRepository;
import kh.edu.istad.platform.iamapplicationservice.result.user.UserResult;
import kh.edu.istad.platform.iamapplicationservice.usecase.user.CreateUserUseCase;
import kh.edu.istad.platform.iamdomaincore.model.User;

import java.time.OffsetDateTime;
import java.util.List;
import java.util.Locale;

public class CreateUserService implements CreateUserUseCase {

    private final UserRepository userRepository;
    private final PasswordHasher passwordHasher;
    private final CurrentActorProvider currentActorProvider;

    public CreateUserService(
            UserRepository userRepository,
            PasswordHasher passwordHasher,
            CurrentActorProvider currentActorProvider
    ) {
        this.userRepository = userRepository;
        this.passwordHasher = passwordHasher;
        this.currentActorProvider = currentActorProvider;
    }

    @Override
    public UserResult execute(CreateUserCommand command) {
        if (command == null) {
            throw new IllegalArgumentException("Create user command is required");
        }

        String username = command.username().trim();
        String email = command.email().trim().toLowerCase(Locale.ROOT);

        if (userRepository.existsByUsername(username)) {
            throw new IllegalArgumentException("Username already exists");
        }

        if (userRepository.existsByEmail(email)) {
            throw new IllegalArgumentException("Email already exists");
        }

        if (command.rawPassword() == null || command.rawPassword().isBlank()) {
            throw new IllegalArgumentException("Password is required");
        }

        String actor = currentActorProvider.getCurrentActor();
        OffsetDateTime now = OffsetDateTime.now();

        User user = new User(
                null, null,
                username, email, false,
                passwordHasher.hash(command.rawPassword()),
                command.givenName(), command.familyName(),
                null, null, null, null, null, null, null, null,
                "en",
                null, false,
                null, null, null, null, null,
                true, true, true, true,
                0, null, null,
                false, null, null,
                now, actor, now, actor,
                0
        );

        User saved = userRepository.save(user);

        return new UserResult(
                saved.uuid(),
                saved.username(),
                saved.email(),
                saved.emailVerified(),
                saved.name(),
                saved.locale(),
                saved.enabled(),
                List.of(),
                saved.createdAt(),
                saved.updatedAt()
        );
    }
}

package kh.edu.istad.platform.iamapplicationservice.service.user;

import kh.edu.istad.platform.iamapplicationservice.port.out.*;
import kh.edu.istad.platform.iamapplicationservice.result.user.UserAuthoritiesResult;
import kh.edu.istad.platform.iamapplicationservice.usecase.user.GetUserAuthoritiesUseCase;
import kh.edu.istad.platform.iamdomaincore.enums.PermissionEffect;
import kh.edu.istad.platform.iamdomaincore.model.Permission;
import kh.edu.istad.platform.iamdomaincore.model.Role;
import kh.edu.istad.platform.iamdomaincore.model.User;

import java.time.OffsetDateTime;
import java.util.*;

public class GetUserAuthoritiesService implements GetUserAuthoritiesUseCase {

    private final UserRepository userRepository;
    private final RoleRepository roleRepository;
    private final PermissionRepository permissionRepository;
    private final UserRoleRepository userRoleRepository;
    private final UserPermissionRepository userPermissionRepository;
    private final RolePermissionRepository rolePermissionRepository;

    public GetUserAuthoritiesService(
            UserRepository userRepository,
            RoleRepository roleRepository,
            PermissionRepository permissionRepository,
            UserRoleRepository userRoleRepository,
            UserPermissionRepository userPermissionRepository,
            RolePermissionRepository rolePermissionRepository
    ) {
        this.userRepository = userRepository;
        this.roleRepository = roleRepository;
        this.permissionRepository = permissionRepository;
        this.userRoleRepository = userRoleRepository;
        this.userPermissionRepository = userPermissionRepository;
        this.rolePermissionRepository = rolePermissionRepository;
    }

    @Override
    public UserAuthoritiesResult execute(UUID userUuid) {
        User user = userRepository.findByUuid(userUuid)
                .orElseThrow(() -> new NoSuchElementException("User not found"));

        OffsetDateTime now = OffsetDateTime.now();

        Map<Long, Role> activeRoles = new LinkedHashMap<>();
        List<UserAuthoritiesResult.RoleAuthority> roleResults = new ArrayList<>();
        Set<String> effective = new TreeSet<>();

        userRoleRepository.findAllByUserId(user.id()).stream()
                .filter(assignment -> assignment.isActiveAt(now))
                .forEach(assignment -> roleRepository.findById(assignment.roleId())
                        .filter(Role::isAssignable)
                        .ifPresent(role -> {
                            activeRoles.put(role.id(), role);
                            roleResults.add(new UserAuthoritiesResult.RoleAuthority(
                                    role.uuid(), role.name(), assignment.expiresAt()
                            ));
                        }));

        for (Role role : activeRoles.values()) {
            rolePermissionRepository.findAllByRoleId(role.id()).stream()
                    .filter(assignment -> assignment.isActiveAt(now))
                    .forEach(assignment -> permissionRepository
                            .findById(assignment.permissionId())
                            .filter(Permission::isAssignable)
                            .ifPresent(permission -> effective.add(permission.name())));
        }

        Map<String, PermissionEffect> directEffects = new LinkedHashMap<>();
        List<UserAuthoritiesResult.DirectPermission> directResults = new ArrayList<>();

        userPermissionRepository.findAllByUserId(user.id()).stream()
                .filter(assignment -> assignment.isActiveAt(now))
                .forEach(assignment -> permissionRepository
                        .findById(assignment.permissionId())
                        .filter(Permission::isAssignable)
                        .ifPresent(permission -> {
                            directEffects.merge(
                                    permission.name(),
                                    assignment.effect(),
                                    (oldEffect, newEffect) ->
                                            oldEffect == PermissionEffect.DENY
                                                    || newEffect == PermissionEffect.DENY
                                                    ? PermissionEffect.DENY
                                                    : PermissionEffect.GRANT
                            );
                        }));

        directEffects.forEach((name, effect) -> {
            directResults.add(
                    new UserAuthoritiesResult.DirectPermission(name, effect)
            );

            if (effect == PermissionEffect.GRANT) {
                effective.add(name);
            } else {
                effective.remove(name);
            }
        });

        return new UserAuthoritiesResult(
                roleResults,
                directResults,
                new ArrayList<>(effective)
        );
    }
}

package kh.edu.istad.platform.iamapplicationservice.service.user;

import kh.edu.istad.platform.iamapplicationservice.port.out.UserRepository;
import kh.edu.istad.platform.iamapplicationservice.result.user.AuthUserResult;
import kh.edu.istad.platform.iamapplicationservice.result.user.UserAuthoritiesResult;
import kh.edu.istad.platform.iamapplicationservice.usecase.user.GetUserAuthoritiesUseCase;
import kh.edu.istad.platform.iamapplicationservice.usecase.user.LoadUserForAuthenticationUseCase;
import kh.edu.istad.platform.iamdomaincore.model.User;

import java.util.ArrayList;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.NoSuchElementException;

public class LoadUserForAuthenticationService
        implements LoadUserForAuthenticationUseCase {

    private final UserRepository userRepository;
    private final GetUserAuthoritiesUseCase getUserAuthoritiesUseCase;

    public LoadUserForAuthenticationService(
            UserRepository userRepository,
            GetUserAuthoritiesUseCase getUserAuthoritiesUseCase
    ) {
        this.userRepository = userRepository;
        this.getUserAuthoritiesUseCase = getUserAuthoritiesUseCase;
    }

    @Override
    public AuthUserResult execute(String usernameOrEmail) {
        if (usernameOrEmail == null || usernameOrEmail.isBlank()) {
            throw new IllegalArgumentException(
                    "Username or email is required"
            );
        }

        User user = userRepository.findByUsernameOrEmail(
                usernameOrEmail.trim()
        ).orElseThrow(() ->
                new NoSuchElementException("User not found"));

        UserAuthoritiesResult userAuthorities =
                getUserAuthoritiesUseCase.execute(user.uuid());

        LinkedHashSet<String> authorities = new LinkedHashSet<>();

        userAuthorities.roles().forEach(role ->
                authorities.add("ROLE_" + role.name()));

        authorities.addAll(userAuthorities.effectivePermissions());

        return new AuthUserResult(
                user.uuid(),
                user.username(),
                user.passwordHash(),
                user.enabled() && !user.deleted(),
                user.accountNonExpired(),
                user.accountNonLocked(),
                user.credentialsNonExpired(),
                new ArrayList<>(authorities)
        );
    }
}

                            : Math.addExact(user.failedLoginAttempts(), 1);

                    OffsetDateTime lastLoginAt = success
                            ? now
                            : user.lastLoginAt();

                    User updatedUser = new User(
                            user.id(), user.uuid(),
                            user.username(), user.email(),
                            user.emailVerified(), user.passwordHash(),
                            user.givenName(), user.familyName(),
                            user.middleName(), user.nickname(),
                            user.gender(), user.birthdate(),
                            user.profileUrl(), user.pictureUrl(),

                            user.locale(),
                            user.phoneNumber(), user.phoneNumberVerified(),
                            user.streetAddress(), user.locality(),
                            user.region(), user.postalCode(),
                            user.country(),
                            user.enabled(),
                            user.accountNonExpired(),
                            user.accountNonLocked(),
                            user.credentialsNonExpired(),
                            failedAttempts,
                            lastLoginAt,
                            user.passwordChangedAt(),
                            user.deleted(),
                            user.deletedAt(),
                            user.deletedBy(),
                            user.createdAt(),
                            user.createdBy(),
                            now,
                            actor,
                            user.version() + 1
                    );

                    userRepository.save(updatedUser);

package kh.edu.istad.platform.iamapplicationservice.service.user;

import kh.edu.istad.platform.iamapplicationservice.command.user.RegisterUserCommand;
import kh.edu.istad.platform.iamapplicationservice.port.out.CurrentActorProvider;

import kh.edu.istad.platform.iamapplicationservice.port.out.UserRepository;
import kh.edu.istad.platform.iamapplicationservice.result.user.UserResult;
import kh.edu.istad.platform.iamapplicationservice.usecase.user.RegisterUserUseCase;
import kh.edu.istad.platform.iamdomaincore.model.User;

import java.time.OffsetDateTime;
import java.util.List;
import java.util.Locale;

public class RegisterUserService implements RegisterUserUseCase {

    private final UserRepository userRepository;
    private final PasswordHasher passwordHasher;
    private final CurrentActorProvider currentActorProvider;

    public RegisterUserService(
            UserRepository userRepository,
            PasswordHasher passwordHasher,
            CurrentActorProvider currentActorProvider
    ) {
        this.userRepository = userRepository;
        this.passwordHasher = passwordHasher;
        this.currentActorProvider = currentActorProvider;
    }

    @Override
    public UserResult execute(RegisterUserCommand command) {

        if (command == null) {
            throw new IllegalArgumentException("Register command is required");
        }

        String username = command.username().trim();
        String email = command.email().trim().toLowerCase(Locale.ROOT);

        if (userRepository.existsByUsername(username)) {
            throw new IllegalArgumentException("Username already exists");
        }

        if (userRepository.existsByEmail(email)) {
            throw new IllegalArgumentException("Email already exists");
        }

        if (command.rawPassword() == null || command.rawPassword().isBlank()) {
            throw new IllegalArgumentException("Password is required");
        }

        String passwordHash = passwordHasher.hash(command.rawPassword());

        if (passwordHash == null || passwordHash.isBlank()) {
            throw new IllegalStateException("Password hashing failed");
        }

        String actor = currentActorProvider.getCurrentActor();

        if (actor == null || actor.isBlank()) {
            throw new IllegalStateException("Current actor is required");
        }

        OffsetDateTime now = OffsetDateTime.now();

        User user = new User(
                null, null, username, email, false, passwordHash,
                command.givenName(), command.familyName(),
                null, null, null, null, null, null, null, null,
                "en", null, false, null, null, null, null, null,
                true, true, true, true,
                0, null, null, false, null, null,
                now, actor, now, actor, 0
        );

        User savedUser = userRepository.save(user);

        return new UserResult(
                savedUser.uuid(),
                savedUser.username(),
                savedUser.email(),
                savedUser.emailVerified(),
                savedUser.name(),
                savedUser.locale(),
                savedUser.enabled(),
                List.of(),
                savedUser.createdAt(),
                savedUser.updatedAt()
        );
    }
}

import kh.edu.istad.platform.iamapplicationservice.port.out.UserRepository;
import kh.edu.istad.platform.iamapplicationservice.usecase.user.ResetUserPasswordUseCase;
import kh.edu.istad.platform.iamdomaincore.model.User;

import java.time.OffsetDateTime;

public class ResetUserPasswordService implements ResetUserPasswordUseCase {

    private final UserRepository repository;
    private final PasswordHasher passwordHasher;
    private final CurrentActorProvider actorProvider;

    public ResetUserPasswordService(
            UserRepository repository,
            PasswordHasher passwordHasher,
            CurrentActorProvider actorProvider
    ) {
        this.repository = repository;
        this.passwordHasher = passwordHasher;
        this.actorProvider = actorProvider;
    }

    @Override
    public void execute(ResetUserPasswordCommand command) {
        if (command == null) {
            throw new IllegalArgumentException("Reset password command is required");
        }

        if (command.newPassword() == null || command.newPassword().isBlank()) {
            throw new IllegalArgumentException("New password is required");
        }

        User user = UserStateSupport.requireActiveUser(
                repository, command.userUuid()
        );

        String passwordHash = passwordHasher.hash(command.newPassword());

        }

        String actor = actorProvider.getCurrentActor();

        if (actor == null || actor.isBlank()) {
            throw new IllegalStateException("Current actor is required");
        }

        repository.save(UserPasswordSupport.updatePassword(
                user, passwordHash, actor, OffsetDateTime.now()
        ));
    }
}
